{\sffamily\bfseries PVG32 valve model with laminar flow on the rod side (\texttt{PVG32\_laminar.m})}

\vspace{4pt}
\begin{lstlisting}
function [Q_A, Q_B, Q_p, u] = PVG32_laminar(param, U,p_h1, p_h2)

% U: user control signal
% p_h1: hydraulic pressure on piston side
% p_h2: hydraulic pressure on piston-rod side
% p_P: pump pressure
% param.p_t: tank pressure

u                          = (param.valve(1).U_N -U)/3;  %(6-U)/3,5.8

if u > param.valve(1).u0

    Q_A = ((u -  param.valve(1).u0)/(1 -  param.valve(1).u0))*param.valve(1).Q0;
    Kv = ((u -  param.valve(1).u0)/(1 -  param.valve(1).u0))*param.valve(1).Kv0;

(*\tikzmark{a1}*)    dp   = p_h2 - param.p_t;
    p_tr = 5e5;                                   % transition pressure [Pa]
    Cv   = param.Cv;
    if abs(dp) <= p_tr
        Q_B = Cv*dp;                   % laminar: linear
    else
        Q_B = Kv*sqrt(abs(dp))*sign(dp);          % turbulent
(*\tikzmark{a2}*)    end

    Q_p = Q_A;
elseif u > -param.valve(1).u0
    Q_B = 1e-10;
    Q_p = Q_B;
    Q_A = 0;
else

(*\tikzmark{b1}*)    dp   = p_h2 - param.p_t;
    p_tr = 7e5;                                   % transition pressure [Pa]
    kv0  = (abs(u) -  param.valve(1).u0)/(1 -  param.valve(1).u0);
    Cv   = param.Cv;
    if abs(dp) <= p_tr
        Q_B = Cv*dp;                   % laminar: linear
    else
        Q_B = -kv0*param.valve(1).Q1;
(*\tikzmark{b2}*)    end

    Kv = (abs(u) -  param.valve(1).u0)/(1 -  param.valve(1).u0)*param.valve(1).Kv1;
    Q_A = -Kv*sqrt(abs(p_h1-param.p_t))*sign(p_h1-param.p_t);

    Q_p = Q_B;
end
\end{lstlisting}

{\sffamily\bfseries\small Laminar coefficient (parameter file)}

\vspace{2pt}
\begin{lstlisting}[numbers=none]
Q1                  = 40/6e4;                                % 40 l/min -> m^3/s
param.Cv            = (Q1) / sqrt(param.valve(2).p_ref);
\end{lstlisting}

\vspace{2pt}
{\small Rod-side flow $Q_\mathrm{B}$ with $\Delta p = p_{\mathrm{h}2} - p_\mathrm{t}$:
\[
Q_\mathrm{B} =
\begin{cases}
C_v\,\Delta p, & |\Delta p| \le p_\mathrm{tr} \quad \text{(laminar)}\\[2pt]
K_v\sqrt{|\Delta p|}\,\operatorname{sign}(\Delta p), & |\Delta p| > p_\mathrm{tr},\ u > u_0 \quad \text{(turbulent)}\\[2pt]
-k_{v0}\,Q_1, & |\Delta p| > p_\mathrm{tr},\ u < -u_0 \quad \text{(flow-controlled)}
\end{cases}
\]
with $C_v = Q_1/\sqrt{p_\mathrm{ref}}$, $Q_1 = 40$\,l/min, and $p_\mathrm{tr} = 5$\,bar (case 1) and $7$\,bar (case 2).}
